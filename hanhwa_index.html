""" dedup.py — Google Sheets 중복 행 제거 스크립트 위대한상상 H열(원문URL) 기준으로 중복 행을 찾아 삭제합니다. - URL이 비어있는 행은 중복으로 처리하지 않음 - 동일 URL 중 가장 위(첫 번째) 행을 유지하고 나머지 삭제 - 실행 전 미리보기로 몇 건 삭제될지 확인 후 진행 실행: python dedup.py """ import gspread import time import config COL_URL = 7 # H열 (0-based) def main(): print(f"\n{'='*55}") print(f" 🧹 중복 행 제거 시작") print(f"{'='*55}\n") # Sheets 연결 gc = gspread.service_account(filename="credentials.json") sh = gc.open_by_url(config.GOOGLE_SHEETS_URL) ws = sh.worksheet("피드") # 전체 데이터 읽기 print(" 데이터 읽는 중...") all_values = ws.get_all_values() if not all_values: print(" 데이터가 없습니다.") return header = all_values[0] rows = all_values[1:] print(f" 총 {len(rows)}행 읽음 (헤더 제외)\n") # 중복 탐지 — URL 기준 seen_urls = set() keep_rows = [] # 유지할 행 번호 (1-based, 헤더 포함) dup_rows = [] # 삭제할 행 번호 (1-based, 헤더 포함) no_url_cnt = 0 for i, row in enumerate(rows): url = row[COL_URL].strip() if len(row) > COL_URL else '' row_num = i + 2 # 1-based + 헤더 if not url: # URL 없는 행은 유지 (중복 판단 불가) keep_rows.append(row_num) no_url_cnt += 1 continue if url in seen_urls: dup_rows.append(row_num) else: seen_urls.add(url) keep_rows.append(row_num) print(f" ── 분석 결과 ──────────────────────────") print(f" 전체 행: {len(rows):,}건") print(f" URL 없는 행: {no_url_cnt:,}건 (유지)") print(f" 중복 행: {len(dup_rows):,}건 (삭제 예정)") print(f" 유지 행: {len(keep_rows):,}건") print(f" ────────────────────────────────────────\n") if not dup_rows: print(" ✅ 중복 행 없음. 종료합니다.") return # 삭제 예정 행 미리보기 (최대 10건) print(f" 삭제 예정 행 미리보기 (최대 10건):") for row_num in dup_rows[:10]: row = all_values[row_num - 1] title = row[4][:40] if len(row) > 4 else '' url = row[COL_URL][:60] if len(row) > COL_URL else '' print(f" 행 {row_num:>5} | {title:<42} | {url}") if len(dup_rows) > 10: print(f" ... 외 {len(dup_rows)-10}건") # 사용자 확인 print(f"\n 총 {len(dup_rows):,}건을 삭제합니다.") confirm = input(" 진행할까요? (yes 입력 후 Enter): ").strip().lower() if confirm != 'yes': print("\n ❌ 취소되었습니다.") return # 삭제 — 연속된 행을 범위로 묶어 batchUpdate 1회 호출 # → 행 번호가 밀리므로 반드시 내림차순 정렬 후 범위 병합 dup_rows_desc = sorted(dup_rows, reverse=True) total = len(dup_rows_desc) print(f"\n {total}건 삭제 중 (범위 병합 방식)...") # 연속 범위로 병합 def to_ranges(rows_desc): """내림차순 행 목록을 연속 범위 리스트로 변환 [(start, end), ...]""" if not rows_desc: return [] ranges = [] start = end = rows_desc[0] for r in rows_desc[1:]: if r == end - 1: end = r else: ranges.append((end, start)) # gspread는 startIndex 포함, endIndex 미포함 start = end = r ranges.append((end, start)) return ranges row_ranges = to_ranges(dup_rows_desc) # batchUpdate 요청 구성 (한 번에 최대 50개 범위씩) BATCH = 50 deleted = 0 for idx in range(0, len(row_ranges), BATCH): batch = row_ranges[idx:idx+BATCH] requests = [ { "deleteDimension": { "range": { "sheetId": ws.id, "dimension": "ROWS", "startIndex": start - 1, # 0-based "endIndex": end, # exclusive } } } for start, end in batch ] ws.spreadsheet.batch_update({"requests": requests}) deleted += sum(end - start + 1 for start, end in batch) print(f" {min(deleted, total)}/{total} 완료") time.sleep(3) # Write quota 여유 print(f"\n{'='*55}") print(f" ✅ 중복 제거 완료!") print(f" 삭제: {total:,}건 / 남은 행: {len(keep_rows):,}건") print(f"{'='*55}\n") if __name__ == "__main__": main()
